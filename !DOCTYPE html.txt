  function selectMode(modeName, btnElement) {
    // หยุดเล่นเสียงทั้งหมดเมื่อเปลี่ยนโหมด
    document.querySelectorAll('audio').forEach(a => a.pause());

    document.querySelectorAll('.content-panel').forEach(p => p.classList.remove('active'));
    document.querySelectorAll('.mode-btn').forEach(b => {
      b.classList.remove('active');
      b.setAttribute('aria-pressed', 'false');
    });

    document.getElementById('panel-' + modeName).classList.add('active');
    btnElement.classList.add('active');
    btnElement.setAttribute('aria-pressed', 'true');
  }

  let currentFontSize = 18;
  function changeFontSize(step) {
    currentFontSize = Math.min(28, Math.max(14, currentFontSize + step));
    document.documentElement.style.setProperty('--base-font-size', currentFontSize + 'px');
  }

  function toggleContrast() {
    document.body.classList.toggle('high-contrast');
  }

  function checkAnswer(selectedIndex) {
    const resultEl = document.getElementById('quiz-result');
    if (selectedIndex === currentTree.correctIndex) {
      resultEl.style.color = '#15803d';
      resultEl.innerText = currentTree.quizSuccessMsg;
    } else {
      resultEl.style.color = '#b91c1c';
      resultEl.innerText = currentTree.quizFailMsg;
    }
